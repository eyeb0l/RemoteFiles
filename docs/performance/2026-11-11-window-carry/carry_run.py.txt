"""Runs `windowcarry` (PerformanceBenchmarkTests/testTransport) against a server that is already up, from a built Mac
checkout or on a phone, and appends its samples to a JSON Lines file.

  carry_run.py mac   --checkout PATH ...   swift test, built with `swift build -c release -Xswiftc -enable-testing --build-tests`
  carry_run.py phone --xctestrun FILE --udid DEVICE ...   xcodebuild test-without-building, built as docs/PERFORMANCE.md says

Common: --host H --port P --user U --key PRIVATE_KEY_FILE --host-key HOST_KEY.pub --root REMOTE_TREE --label L --output F
        [--runs N] [--arms growing,carry,...] [--idle-ms N] [--only windowcarry]
"""
import argparse, os, pathlib, subprocess, sys

parser = argparse.ArgumentParser()
parser.add_argument("where", choices=("mac", "phone"))
for name in ("host", "port", "user", "key", "host-key", "root", "label", "output"):
    parser.add_argument("--" + name, required=True)
parser.add_argument("--runs", default="2")
parser.add_argument("--arms", default="growing,carry,carryhold,fixed64")
parser.add_argument("--idle-ms", default="0")
parser.add_argument("--only", default="windowcarry")
parser.add_argument("--checkout")
parser.add_argument("--xctestrun")
parser.add_argument("--udid")
parser.add_argument("--timeout", type=int, default=3600)
args = parser.parse_args()

prefix = "TEST_RUNNER_" if args.where == "phone" else ""
values = dict(REMOTEFILES_PERFORMANCE_BENCHMARK="1", REMOTEFILES_BENCHMARK_RUNS=args.runs, REMOTEFILES_BENCHMARK_ROOT=args.root,
              REMOTEFILES_BENCHMARK_LABEL=args.label, REMOTEFILES_BENCHMARK_ONLY=args.only,
              REMOTEFILES_BENCHMARK_CARRY_ARMS=args.arms, REMOTEFILES_BENCHMARK_CARRY_IDLE_MS=args.idle_ms,
              REMOTEFILES_BENCHMARK_PRIVATE_KEY=pathlib.Path(args.key).read_text(),
              REMOTEFILES_BENCHMARK_HOST_KEY=pathlib.Path(args.host_key).read_text(),
              REMOTEFILES_SFTP_HOST=args.host, REMOTEFILES_SFTP_PORT=args.port, REMOTEFILES_SFTP_USER=args.user)
env = dict(os.environ)
env.update({prefix + key: value for key, value in values.items()})
output = str(pathlib.Path(args.output).resolve())
if args.where == "mac":
    env["REMOTEFILES_BENCHMARK_OUTPUT"] = output
    before = sum(1 for _ in open(output)) if os.path.exists(output) else 0
    result = subprocess.run(["swift", "test", "-c", "release", "-Xswiftc", "-enable-testing", "--skip-build", "--filter",
                             "PerformanceBenchmarkTests/testTransport"], cwd=args.checkout, env=env, capture_output=True, text=True, timeout=args.timeout)
    count = (sum(1 for _ in open(output)) if os.path.exists(output) else 0) - before
else:
    result = subprocess.run(["xcodebuild", "test-without-building", "-xctestrun", args.xctestrun, "-destination", f"id={args.udid}",
                             "-only-testing:RemoteFilesTests/PerformanceBenchmarkTests/testTransport", "-parallel-testing-enabled", "NO",
                             "-collect-test-diagnostics", "never"], env=env, capture_output=True, text=True, timeout=args.timeout)
    lines = [line.split("BENCHJSON ", 1)[1] for line in result.stdout.splitlines() if "BENCHJSON " in line]
    with open(output, "a") as handle:
        handle.writelines(line + "\n" for line in lines)
    count = len(lines)
print(f"{args.label}: exit {result.returncode}, {count} scenarios", flush=True)
if result.returncode:
    text = result.stdout + result.stderr
    print("\n".join(line for line in text.splitlines() if " error" in line.lower() or "locked" in line.lower() or "failed" in line.lower())[:3000])
    sys.exit(1)
