# RemoteFiles read-only search helper, protocol 2. Executed in memory with Python -I -S -B.
# Writes one JSON object a line as it goes: what it has found since the last line, with running
# totals. The last line has final set, and only that one can carry an error.
import base64, collections, errno, json, os, re, signal, stat, sys, time

r = json.loads(base64.b64decode(sys.argv[1]))
root = r['root']
result = dict(version=2, root=root, candidates=[], folders=0, unreadableFolders=0,
              deepFolders=0, files=0, skippedFiles=0, unreadableFiles=0, bytes=0,
              charged=0, stop=None, error=None, final=False)
started = time.monotonic()
output_bytes = 0
unsent_bytes = 0
last_line = None
last_folders = 0
entries_seen = 0
expired = False
scanning = True
root_fd = None

class Unsupported(Exception): pass
class ResourceLimit(Exception): pass
class Expired(Exception): pass

def alarm(signum, frame):
    global expired
    if scanning and not expired:
        expired = True
        result['stop'] = 'timeLimit'
        raise Expired()

def check_time():
    if expired or time.monotonic() - started >= r['seconds']:
        alarm(None, None)
        raise Expired()

def open_directory(parts, absolute=False):
    # Pin the verified root, then resolve descendants relative to that descriptor.
    fd = os.open('/', os.O_RDONLY | os.O_DIRECTORY) if absolute else os.dup(root_fd)
    try:
        for part in parts:
            next_fd = os.open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
            os.close(fd)
            fd = next_fd
        return fd
    except BaseException:
        # A deadline may arrive just after close completed. Preserve that exception instead of
        # replacing it with EBADF from cleanup (the next descriptor is reclaimed at process exit).
        try:
            os.close(fd)
        except OSError:
            pass
        raise

def write_line(value):
    # A deadline in the middle of a write would cut the line in half. check_time sees it afterwards.
    global scanning
    scanning = False
    try:
        sys.stdout.write(json.dumps(value, ensure_ascii=True, separators=(',', ':')) + '\n')
        sys.stdout.flush()
    finally:
        scanning = True

def send_progress():
    # Called after each folder, so a folder's matches arrive together and the app orders them as it would
    # have. The first results go at once, then ten lines a second at most, or sooner if they pile up.
    global last_line, last_folders, unsent_bytes
    now = time.monotonic()
    if not result['candidates'] and result['folders'] == last_folders:
        return
    if last_line is not None and now - last_line < 0.1 and unsent_bytes < 65536:
        return
    if last_line is None and not result['candidates']:
        return
    write_line(result)
    last_line, last_folders, unsent_bytes = now, result['folders'], 0
    result['candidates'] = []

query_lower = r['query'].lower()
query_ascii = r['query'].isascii()
query_bytes = query_lower.encode('ascii') if query_ascii else b''

def candidate(name, data):
    # Unicode is always a candidate: the app applies its exact Foundation matching semantics.
    if not query_ascii or not name.isascii():
        return True
    if data is None:
        return query_lower in name.lower()
    if not data.isascii():
        return True
    return query_bytes in data.lower()

def surrogate(name):
    # Unrepresentable filesystem bytes cannot be safely mapped to Swift paths.
    return not name.isascii() and any(0xD800 <= ord(c) <= 0xDFFF for c in name)

candidates_sent = 0
# translate deletes these and leaves only the control bytes that make a file binary.
printable = bytes(byte for byte in range(256) if not ((byte < 32 and byte not in (9, 10, 13)) or byte == 127))

try:
    import unicodedata
except ImportError:
    unicodedata = None

def natural(name):
    # Close to the app's localized order: digits by value, then accents and case set aside, so that
    # folders reach the app in nearly the order it would have walked them. It still decides the order.
    folded = name
    if unicodedata is not None and not name.isascii():
        folded = ''.join(c for c in unicodedata.normalize('NFKD', name) if not unicodedata.combining(c))
    return ([(0, int(p)) if p.isdigit() else (1, p.casefold()) for p in re.split(r'([0-9]+)', folded)], name)

def add(path, kind, info, data=None):
    global output_bytes, unsent_bytes, candidates_sent
    item = dict(path=path, kind=kind, size=info.st_size, modified=info.st_mtime)
    if data is not None:
        item['data'] = base64.b64encode(data).decode('ascii')
    size = len(json.dumps(item)) + 2
    output_bytes += size
    unsent_bytes += size
    candidates_sent += 1
    if candidates_sent > 10000 or output_bytes > 23 * 1024 * 1024:
        raise ResourceLimit()
    result['candidates'].append(item)

def scan_file(fd, name, path):
    suffix = name.lower().rsplit('.', 1)[-1]
    text = suffix in r['textExtensions'] or suffix in ('md', 'markdown') or '.' not in name or name.startswith('.')
    if suffix in r['excludedExtensions'] or not text:
        result['skippedFiles'] += 1
        return
    try:
        info = os.stat(name, dir_fd=fd, follow_symlinks=False)
    except OSError:
        return
    if info.st_size > r['fileBytes']:
        result['skippedFiles'] += 1
        return
    if result['files'] >= r['files']:
        result['stop'] = 'fileLimit'
        return
    remaining = r['totalBytes'] - result['charged']
    if remaining <= 1:
        result['stop'] = 'byteLimit'
        return
    limit = min(r['fileBytes'], remaining - 1)
    result['files'] += 1
    result['charged'] += limit + 1
    file_fd = None
    try:
        check_time()
        # O_NONBLOCK prevents a file replaced by a FIFO from hanging the job. fstat rejects it.
        file_fd = os.open(name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=fd)
        if not stat.S_ISREG(os.fstat(file_fd).st_mode):
            result['skippedFiles'] += 1
            return
        chunks = []
        count = 0
        while count <= limit:
            check_time()
            chunk = os.read(file_fd, min(65536, limit + 1 - count))
            if not chunk:
                break
            chunks.append(chunk)
            count += len(chunk)
        if count > limit:
            if limit < r['fileBytes']:
                result['stop'] = 'byteLimit'
            else:
                result['skippedFiles'] += 1
            return
        data = b''.join(chunks)
        result['charged'] -= limit + 1 - count
        result['bytes'] += count
        if data.translate(None, printable):
            result['skippedFiles'] += 1
            return
        try:
            data.decode('utf-8')
        except UnicodeDecodeError:
            result['skippedFiles'] += 1
            return
        if data and candidate(name, data):
            add(path, 'file', info, data)
    except OSError:
        result['unreadableFiles'] += 1
    finally:
        if file_fd is not None:
            os.close(file_fd)

try:
    # POSIX descriptor-relative operations are required. Windows/older runtimes fall back to SFTP.
    if sys.version_info < (3, 7) or os.open not in os.supports_dir_fd or os.stat not in os.supports_dir_fd or os.scandir not in os.supports_fd:
        raise Unsupported()
    if not hasattr(os, 'O_NOFOLLOW') or not hasattr(signal, 'setitimer'):
        raise Unsupported()
    signal.signal(signal.SIGALRM, alarm)
    signal.setitimer(signal.ITIMER_REAL, r['seconds'])
    identity = r.get('identity')
    handle = r.get('rootHandle', '')
    try:
        if identity is not None:
            # Compare kernel identities before listing/reading user data. SFTP keeps its root
            # open throughout, preventing inode reuse. No cross-process inspection is needed.
            if os.readlink('/proc/self/ns/mnt') != identity['namespace']:
                raise Unsupported()
            root_fd = open_directory(root.strip('/').split('/') if root != '/' else [], absolute=True)
            with open('/proc/self/fdinfo/' + str(root_fd), encoding='ascii') as info:
                fields = dict(line.split(':', 1) for line in info.read(4097).splitlines() if ':' in line)
            if os.fstat(root_fd).st_ino != identity['inode'] or int(fields['mnt_id']) != identity['mountID']:
                raise Unsupported()
        elif re.fullmatch(r'/dev/fd/[0-9]+', handle):
            # Also usable by callers that already hold the directory in this process.
            root_fd = os.dup(int(handle.rsplit('/', 1)[-1]))
            if not stat.S_ISDIR(os.fstat(root_fd).st_mode):
                raise Unsupported()
        else:
            raise Unsupported()
    except (OSError, KeyError, ValueError):
        raise Unsupported()
    check_time()
    if root_fd is not None:
        queue = collections.deque([(root, 0)])
        while queue and not result['stop']:
            check_time()
            if result['folders'] + result['unreadableFolders'] >= r['folders']:
                result['stop'] = 'folderLimit'
                break
            path, depth = queue.popleft()
            fd = None
            try:
                relative = path[len(root):].strip('/')
                fd = open_directory(relative.split('/') if relative else [])
                entries = []
                with os.scandir(fd) as listing:
                    for entry in listing:
                        check_time()
                        entries_seen += 1
                        if entries_seen > 100000:
                            raise ResourceLimit()
                        if r['hidden'] or not entry.name.startswith('.'):
                            entries.append(entry)
                result['folders'] += 1
                subfolders = []
                if r.get('files') is None:
                    # The directory already says what each entry is, so a name that doesn't match takes
                    # no system call. On a cold disk each stat can be a read. The app orders the matches.
                    for entry in entries:
                        name = entry.name
                        if surrogate(name):
                            raise Unsupported()
                        try:
                            if entry.is_dir(follow_symlinks=False):
                                subfolders.append(name)
                            if candidate(name, None):
                                info = os.stat(name, dir_fd=fd, follow_symlinks=False)
                                mode = info.st_mode
                                kind = 'directory' if stat.S_ISDIR(mode) else 'file' if stat.S_ISREG(mode) else 'symlink' if stat.S_ISLNK(mode) else 'other'
                                add(os.path.join(path, name), kind, info)
                        except OSError:
                            continue
                else:
                    # Which files fit the limits depends on their order.
                    entries.sort(key=lambda entry: natural(entry.name))
                    for entry in entries:
                        name = entry.name
                        if surrogate(name):
                            raise Unsupported()
                        check_time()
                        try:
                            if entry.is_dir(follow_symlinks=False):
                                subfolders.append(name)
                            elif entry.is_file(follow_symlinks=False):
                                scan_file(fd, name, os.path.join(path, name))
                            else:
                                result['skippedFiles'] += 1
                        except OSError:
                            continue
                        if result['stop']:
                            break
                subfolders.sort(key=natural)
                for name in subfolders:
                    if depth < r['depth']:
                        # Do not retain an unbounded queue even for a very wide directory.
                        if len(queue) < r['folders']:
                            queue.append((os.path.join(path, name), depth + 1))
                    else:
                        result['deepFolders'] += 1
                send_progress()
            except OSError as error:
                if depth == 0:
                    result['error'] = 'permissionDenied' if error.errno == errno.EACCES else 'notFound' if error.errno == errno.ENOENT else 'outsideRoot'
                    break
                result['unreadableFolders'] += 1
            finally:
                if fd is not None:
                    os.close(fd)
except Expired:
    result['stop'] = 'timeLimit'
except Unsupported:
    result['error'] = 'unsupported'
except ResourceLimit:
    result['error'] = 'resourceLimit'
except BrokenPipeError:
    os._exit(0) # The app stopped listening, so there's no one to finish for.
except (AttributeError, NotImplementedError):
    result['error'] = 'unsupported'
finally:
    try:
        scanning = False
        if root_fd is not None:
            os.close(root_fd)
        if hasattr(signal, 'setitimer'):
            signal.setitimer(signal.ITIMER_REAL, 0)
    except Expired:
        scanning = False
        result['stop'] = 'timeLimit'

if result['error']:
    result['candidates'] = []
result['final'] = True
try:
    write_line(result)
except BrokenPipeError:
    os._exit(0) # The app stopped listening.
